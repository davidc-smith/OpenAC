```csharp
using AcDream.Plugin.Abstractions;

namespace AcDream.Plugins.ControlsDemo;

/// <summary>
/// One window per group of plugin markup controls, each with a status line
/// that says what the last callback received: Inputs (checkboxes, number
/// fields, radio groups, separators), Keys and colours (hotkey boxes, colour
/// pickers), Layout (a split holding expanders), Table (a list with a header,
/// sorting, resizable columns, multi-select and a context menu), Tree (a tree
/// list) and Dialogs (message boxes and a modal editor). Every window opts in
/// to the plugin theme, so switching Plugin appearance from the shelf's
/// right-click menu shows each control in Classic and in both modern themes.
/// </summary>
public sealed class ControlsDemoPlugin : IAcDreamPlugin
{
    private IPluginHost? _host;
    private readonly List<IDisposable> _windows = [];

    public void Initialize(IPluginHost host) => _host = host;

    public void Enable()
    {
        if (_host is not { } host) return;
        // The host loads plugin assemblies from memory, so Assembly.Location is
        // empty; a relative markup path is read from the install folder.
        string Markup(string file) => host.PluginDirectory is { } dir ? Path.Combine(dir, file) : file;
        void Add(PluginPanelDescriptor descriptor, string file, object binding) =>
            _windows.Add(host.Ui.RegisterPanel(descriptor, Markup(file), binding));

        Add(new PluginPanelDescriptor("inputs", "Controls: Inputs") { IconText = "CI" }, "inputs.xml", new InputsBinding());
        // Later windows are added here.
    }

    public void Disable()
    {
        foreach (IDisposable window in _windows) window.Dispose();
        _windows.Clear();
    }
}
```

```csharp
namespace AcDream.Plugins.ControlsDemo;

/// <summary>
/// The Inputs window: three tick boxes under a select-all box that shows a
/// dash while they disagree, a box that only displays, two number fields (a
/// whole number and one with a decimal place), and a radio group beside a
/// segmented bar.
/// </summary>
public sealed class InputsBinding
{
    private static readonly string[] Modes = ["Melee", "Missile", "Magic"];

    public bool Rares { get; private set; } = true;
    public Action<bool> SetRares => value => { Rares = value; Report("Rares", value); };
    public bool Scrolls { get; private set; }
    public Action<bool> SetScrolls => value => { Scrolls = value; Report("Scrolls", value); };
    public bool Salvage { get; private set; } = true;
    public Action<bool> SetSalvage => value => { Salvage = value; Report("Salvage", value); };

    /// <summary>Ticked when all three are, clear when none are, a dash otherwise.</summary>
    public bool? AllPickups => Rares && Scrolls && Salvage ? true
        : !Rares && !Scrolls && !Salvage ? false
        : null;

    /// <summary>The value every pickup should take: true from clear or a dash, false from ticked.</summary>
    public Action<bool> SetAllPickups => value =>
    {
        Rares = Scrolls = Salvage = value;
        Status = value ? "Picking up everything" : "Picking up nothing";
    };

    /// <summary>Bound to a disabled box with no onchange: it only displays.</summary>
    public bool Pyreals => true;

    public int HealAt { get; private set; } = 60;
    public Action<int> SetHealAt => value => { HealAt = value; Status = $"Heal at {value}%"; };
    public double Range { get; private set; } = 12.5;
    public Action<double> SetRange => value => { Range = value; Status = $"Range {value:0.0} m"; };

    public int Mode { get; private set; }
    public Action<int> SetMode => index => { Mode = index; Status = $"{Modes[index]} mode"; };
    public string ModeHint => Mode switch { 0 => "Swords and fists", 1 => "Bows and atlatls", _ => "War and life" };

    public IReadOnlyList<string> Speeds { get; } = ["Walk", "Jog", "Run"];
    public int Speed { get; private set; } = 2;
    public Action<int> SetSpeed => index => { Speed = index; Status = $"{Speeds[index]} speed"; };

    public string Status { get; private set; } = "Change anything";

    private void Report(string what, bool on) => Status = $"{what} {(on ? "on" : "off")}";
}
```

```csharp
using AcDream.App.UI;
using AcDream.Plugin.Abstractions;
using AcDream.Plugins.ControlsDemo;

namespace AcDream.App.Tests.UI;

/// <summary>
/// The ControlsDemo sample's windows build against its real bindings and draw
/// in every plugin theme, with their controls inside the window, and the
/// interactions each window demonstrates reach the binding as the docs say.
/// </summary>
public sealed class ControlsDemoSampleTests
{
    public static TheoryData<PluginUiTheme> Themes => new() { PluginUiTheme.Classic, PluginUiTheme.Moss, PluginUiTheme.Brass };

    private static string Markup(string file)
    {
        string? directory = AppContext.BaseDirectory;
        while (directory is not null && !File.Exists(Path.Combine(directory, "AcDream.slnx")))
            directory = Directory.GetParent(directory)?.FullName;
        Assert.NotNull(directory);
        return File.ReadAllText(Path.Combine(directory, "samples", "AcDream.Plugins.ControlsDemo", file));
    }

    private static (UiRoot Root, UiPanel Panel) Mount(string file, object binding, PluginUiTheme theme = PluginUiTheme.Classic)
    {
        UiPanel panel = MarkupDocument.Build(Markup(file), binding, id => (id, 16, 16),
            themes: new PluginUiThemeSettings { Theme = theme });
        var root = new UiRoot { Width = 1600, Height = 900 };
        root.AddChild(panel);
        root.Tick(0, 0);
        MarkupDrawProbe.Draw(root, 1600, 900);
        return (root, panel);
    }

    /// <summary>Every child starts inside the window; all but labels (sized by their text) end inside it.</summary>
    private static void AssertInside(UiPanel panel)
    {
        foreach (UiElement child in panel.Children)
        {
            Assert.True(child.Left >= 0f && child.Top >= 0f, $"{child.GetType().Name} starts outside the window");
            if (child is UiLabel) continue;
            Assert.True(child.Left + child.Width <= panel.Width, $"{child.GetType().Name} overflows the width");
            Assert.True(child.Top + child.Height <= panel.Height, $"{child.GetType().Name} overflows the height");
        }
    }

    /// <summary>A left click at a point given in the window's own coordinates.</summary>
    private static void Click(UiRoot root, UiPanel panel, float x, float y)
    {
        int sx = (int)(panel.Left + x), sy = (int)(panel.Top + y);
        root.OnMouseDown(UiMouseButton.Left, sx, sy);
        root.OnMouseUp(UiMouseButton.Left, sx, sy);
        root.Tick(0, 0);
    }

    [Theory]
    [MemberData(nameof(Themes))]
    public void Inputs_builds_and_lays_out(PluginUiTheme theme)
    {
        var (_, panel) = Mount("inputs.xml", new InputsBinding(), theme);
        AssertInside(panel);
    }

    [Fact]
    public void Inputs_select_all_ticks_every_pickup_and_shows_a_dash_when_they_disagree()
    {
        var binding = new InputsBinding();
        var (root, panel) = Mount("inputs.xml", binding);
        Assert.Null(binding.AllPickups);

        Click(root, panel, 20, 54);   // "Everything below"
        Assert.True(binding.Rares && binding.Scrolls && binding.Salvage);
        Assert.Equal(true, binding.AllPickups);

        Click(root, panel, 38, 74);   // "Rares"
        Assert.False(binding.Rares);
        Assert.Null(binding.AllPickups);
        Assert.Equal("Rares off", binding.Status);
    }
}
```

```csharp
using AcDream.Plugin.Abstractions;

namespace AcDream.Plugins.ControlsDemo;

/// <summary>
/// The Keys and colours window: two hotkey boxes, with a note saying whether
/// each chord could be registered as a plugin hotkey, and two colour pickers
/// whose colours paint a bar and a heading beside them.
/// </summary>
public sealed class PickersBinding
{
    public string RadarKey { get; private set; } = "Ctrl+R";
    public Action<string> SetRadarKey => value => { RadarKey = value; Status = $"Radar key {Describe(value)}"; };
    public string PauseKey { get; private set; } = string.Empty;
    public Action<string> SetPauseKey => value => { PauseKey = value; Status = $"Pause key {Describe(value)}"; };

    /// <summary>A box can record keys (F13, Caps Lock) that a plugin hotkey can't use.</summary>
    public string KeyNote => $"Hotkeys: radar {Usable(RadarKey)}, pause {Usable(PauseKey)}";

    /// <summary>0xAARRGGBB, with the picker's alpha strip.</summary>
    public uint BarColor { get; private set; } = 0xC040A040;
    public Action<uint> SetBarColor => value => { BarColor = value; Status = $"Bar #{value:X8}"; };

    /// <summary>#AARRGGBB; the picker has no alpha strip, so it stays opaque.</summary>
    public string HeadingColor { get; private set; } = "#FFE8C060";
    public Action<string> SetHeadingColor => value => { HeadingColor = value; Status = $"Heading {value}"; };

    public string Status { get; private set; } = "Click a box or a swatch";

    private static string Describe(string chord) => chord.Length == 0 ? "cleared" : $"set to {chord}";

    private static string Usable(string chord) =>
        chord.Length == 0 ? "unbound"
        : PluginKeyChord.TryParseHotkeyText(chord, out _) ? "usable"
        : "not usable";
}
```

```csharp
using System.Globalization;
using AcDream.Plugin.Abstractions;

namespace AcDream.Plugins.ControlsDemo;

/// <summary>
/// The Table window: loot rows with a select-all keep column, sortable name,
/// value and burden columns (burden sorts by its number, not its "50 bu"
/// text), a Use button per row, multi-select, and a right-click menu. Every
/// row index the list hands back is the plugin's own, however it is sorted.
/// Removing rows asks first, in a box that blocks only this window.
/// </summary>
public sealed class TableBinding
{
    private sealed class Item(string name, int value, int burden)
    {
        public string Name { get; } = name;
        public int Value { get; } = value;
        public int Burden { get; } = burden;
        public bool Keep { get; set; }
    }

    private readonly IUiRegistry _ui;
    private readonly List<Item> _items =
    [
        new("Mana Stone", 3000, 50), new("Steel Salvage", 520, 100),
        new("Scroll of Blade Blessing", 1200, 30), new("Peridot", 400, 10),
        new("Ornate Bracelet", 2600, 40), new("Lockpick", 25, 5),
        new("Healing Kit", 180, 25), new("Pyreal Nugget", 75, 15),
    ];
    private int _found;

    public TableBinding(IUiRegistry ui)
    {
        _ui = ui;
        Rebuild();
    }

    public IReadOnlyList<bool> Keep { get; private set; } = [];
    public IReadOnlyList<string> Names { get; private set; } = [];
    public IReadOnlyList<string> Values { get; private set; } = [];
    public IReadOnlyList<string> Burdens { get; private set; } = [];
    public IReadOnlyList<double> BurdenKeys { get; private set; } = [];
    public IReadOnlyList<string> Actions { get; private set; } = [];

    /// <summary>The selected rows, as the plugin's own indices in ascending order.</summary>
    public IReadOnlyList<int> Picked { get; private set; } = [];
    public Action<IReadOnlyList<int>> SetPicked => rows => { Picked = rows; Status = $"{rows.Count} selected"; };
    public bool HasSelection => Picked.Count > 0;

    public Action<int> ToggleKeep => row =>
    {
        _items[row].Keep = !_items[row].Keep;
        Rebuild();
        Status = $"{(_items[row].Keep ? "Keeping" : "Not keeping")} {_items[row].Name}";
    };

    /// <summary>The value every row should take, from the select-all header.</summary>
    public Action<bool> KeepAll => keep =>
    {
        foreach (Item item in _items) item.Keep = keep;
        Rebuild();
        Status = keep ? "Keeping everything" : "Keeping nothing";
    };

    public Action<int> Act => row => Status = $"Used {_items[row].Name}";

    /// <summary>The right-clicked row (already selected by the list), or -1 over the header.</summary>
    public Action<int> MenuOpened => row => Status = row < 0 ? "Menu over the header" : $"Menu on {_items[row].Name}";

    /// <summary>Items are Keep|Don't keep|-|Remove...; the index counts the separator.</summary>
    public Action<int> MenuPicked => index =>
    {
        switch (index)
        {
            case 0: SetKeep(true); break;
            case 1: SetKeep(false); break;
            case 3: RemoveSelected(); break;
        }
    };

    public Action AddRow => () =>
    {
        _found++;
        _items.Add(new Item($"Found item {_found}", 10 * _found, _found));
        Rebuild();
        Status = $"Added Found item {_found}";
    };

    public Action RemoveSelected => () =>
    {
        if (Picked.Count == 0) return;
        int[] rows = [.. Picked];
        string what = rows.Length == 1 ? _items[rows[0]].Name : $"{rows.Length} items";
        // The box blocks this window, so the rows can't change before the answer.
        bool shown = _ui.Confirm(
            new PluginDialog("Remove items", $"Remove {what} from the table?", ["Remove", "Keep"], OwnerView: "table"),
            answer =>
            {
                if (answer != 0) { Status = "Nothing removed"; return; }
                foreach (int row in rows.OrderDescending()) _items.RemoveAt(row);
                Picked = [];
                Rebuild();
                Status = $"Removed {what}";
            });
        if (!shown) Status = "This host has no dialogs";
    };

    public string Status { get; private set; } = "Click, Ctrl-click or Shift-click rows";

    private void SetKeep(bool keep)
    {
        foreach (int row in Picked) _items[row].Keep = keep;
        Rebuild();
        Status = $"{(keep ? "Keeping" : "Not keeping")} {Picked.Count} rows";
    }

    private void Rebuild()
    {
        Keep = _items.Select(i => i.Keep).ToArray();
        Names = _items.Select(i => i.Name).ToArray();
        Values = _items.Select(i => i.Value.ToString(CultureInfo.InvariantCulture)).ToArray();
        Burdens = _items.Select(i => $"{i.Burden} bu").ToArray();
        BurdenKeys = _items.Select(i => (double)i.Burden).ToArray();
        Actions = _items.Select(_ => "Use").ToArray();
    }
}
```

```csharp
/// <summary>
/// The Tree window: rules flattened parent-first with a depth and a stable
/// key per row. The client keeps (and remembers) which rows are expanded; the
/// plugin is only told. A right click turns a whole branch on or off.
/// </summary>
public sealed class TreeBinding
{
    private sealed class Rule(string key, int level, string name, string value)
    {
        public string Key { get; } = key;
        public int Level { get; } = level;
        public string Name { get; } = name;
        public string Value { get; } = value;
        public bool On { get; set; } = true;
    }

    private readonly Rule[] _rules =
    [
        new("loot", 0, "Looting", ""),
        new("loot.salvage", 1, "Salvage", "workmanship 6+"),
        new("loot.salvage.steel", 2, "Steel", "always"),
        new("loot.salvage.iron", 2, "Iron", "imbue only"),
        new("loot.scrolls", 1, "Scrolls", ""),
        new("loot.scrolls.7", 2, "Level 7", "unknown only"),
        new("loot.scrolls.8", 2, "Level 8", "all"),
        new("combat", 0, "Combat", ""),
        new("combat.buffs", 1, "Buffs", "rebuff at 5 min"),
        new("combat.targets", 1, "Targets", ""),
        new("combat.targets.drudge", 2, "Drudges", "priority 2"),
        new("combat.targets.olthoi", 2, "Olthoi", "priority 1"),
        new("nav", 0, "Navigation", ""),
        new("nav.routes", 1, "Routes", "3 saved"),
    ];
    private int _menuRow = -1;

    public TreeBinding()
    {
        Levels = _rules.Select(r => r.Level).ToArray();
        Keys = _rules.Select(r => r.Key).ToArray();
        Names = _rules.Select(r => r.Name).ToArray();
        Values = _rules.Select(r => r.Value).ToArray();
        Refresh();
    }

    public IReadOnlyList<int> Levels { get; }
    public IReadOnlyList<string> Keys { get; }
    public IReadOnlyList<string> Names { get; }
    public IReadOnlyList<string> Values { get; }
    public IReadOnlyList<bool> On { get; private set; } = [];

    public int Selected { get; private set; } = -1;
    public Action<int> Select => row => { Selected = row; Status = $"{_rules[row].Name} selected"; };

    public Action<int> Toggle => row =>
    {
        _rules[row].On = !_rules[row].On;
        Refresh();
        Status = $"{_rules[row].Name} {(_rules[row].On ? "on" : "off")}";
    };

    /// <summary>Told each expand or collapse; the client applies it.</summary>
    public Action<string, bool> Expanded => (key, open) =>
        Status = $"{_rules.First(r => r.Key == key).Name} {(open ? "expanded" : "collapsed")}";

    public Action<int> MenuOpened => row => _menuRow = row;

    /// <summary>Items are Enable branch|Disable branch.</summary>
    public Action<int> MenuPicked => index =>
    {
        if (_menuRow < 0) return;
        bool on = index == 0;
        int level = _rules[_menuRow].Level;
        _rules[_menuRow].On = on;
        for (int row = _menuRow + 1; row < _rules.Length && _rules[row].Level > level; row++)
            _rules[row].On = on;
        Refresh();
        Status = $"{_rules[_menuRow].Name} branch {(on ? "on" : "off")}";
    };

    public string Status { get; private set; } = "Click an arrow or right-click a rule";

    private void Refresh() => On = _rules.Select(r => r.On).ToArray();
}
```

```csharp
using AcDream.Plugin.Abstractions;

namespace AcDream.Plugins.ControlsDemo;

/// <summary>
/// The modal "Edit profile" window. It edits a copy, so Cancel and the close
/// button change nothing; Save copies it back. Hiding the window ends the
/// block on the Dialogs window.
/// </summary>
public sealed class EditorBinding(IUiRegistry ui, Profile profile)
{
    public string Name { get; private set; } = profile.Name;
    public Action<string> SetName => value => Name = value;
    public int Level { get; private set; } = profile.Level;
    public Action<int> SetLevel => value => Level = value;
    public bool UseOnLogin { get; private set; } = profile.UseOnLogin;
    public Action<bool> SetUseOnLogin => value => UseOnLogin = value;

    /// <summary>Starts a fresh copy; called before the window is shown.</summary>
    public void Load()
    {
        Name = profile.Name;
        Level = profile.Level;
        UseOnLogin = profile.UseOnLogin;
    }

    public Action Save => () =>
    {
        profile.Name = Name.Trim().Length == 0 ? profile.Name : Name.Trim();
        profile.Level = Level;
        profile.UseOnLogin = UseOnLogin;
        ui.HidePanel("editor");
    };

    public Action Cancel => () => ui.HidePanel("editor");
}
```

`samples/AcDream.Plugins.ControlsDemo/DialogsBinding.cs`:

```csharp
using AcDream.Plugin.Abstractions;

namespace AcDream.Plugins.ControlsDemo;

/// <summary>
/// The Dialogs window: message boxes with two, three and four buttons, one
/// with no owner (centred on the screen, blocking nothing), two queued for
/// this window, and a modal editor. The last button is always the safe one,
/// because Escape, the close button and unloading answer with it.
/// </summary>
public sealed class DialogsBinding(IUiRegistry ui, Profile profile, EditorBinding editor)
{
    private const string Owner = "dialogs";

    public Action AskDefault => () => Ask(new PluginDialog(
        "Two buttons", "No buttons given means OK and Cancel. Escape answers Cancel.", OwnerView: Owner));

    public Action AskThree => () => Ask(new PluginDialog(
        "Unsaved changes", "Save the route before closing it?", ["Save", "Discard", "Cancel"], Owner));

    public Action AskFour => () => Ask(new PluginDialog(
        "Pick a pace", "Four buttons widen the box to fit them.", ["Walk", "Jog", "Run", "Stay"], Owner));

    public Action AskUnowned => () => Ask(new PluginDialog(
        "No owner", "Without an owner window the box is centred on the screen and blocks nothing.", ["Got it"]));

    public Action AskTwo => () =>
    {
        Ask(new PluginDialog("First of two", "The second box waits until this one is answered.", ["Next"], Owner));
        Ask(new PluginDialog("Second of two", "Boxes for one window show one at a time.", ["Done"], Owner));
    };

    public Action EditProfile => () =>
    {
        editor.Load();
        Status = ui.ShowModal("editor", Owner) ? "Editing the profile" : "No editor window";
    };

    /// <summary>Items are Edit profile...|Reset profile.</summary>
    public Action<int> ProfileMenu => index =>
    {
        if (index == 0) { EditProfile(); return; }
        Ask(new PluginDialog("Reset profile", $"Reset \"{profile.Name}\" to its defaults?", ["Reset", "Keep"], Owner),
            answer => { if (answer == 0) profile.Reset(); });
    };

    public string ProfileSummary => $"{profile.Name}, level {profile.Level}{(profile.UseOnLogin ? ", on login" : "")}";

    public string Status { get; private set; } = "Press a button";

    private void Ask(PluginDialog dialog, Action<int>? then = null)
    {
        bool shown = ui.Confirm(dialog, index =>
        {
            Status = $"{dialog.Title}: {dialog.EffectiveButtons[index]}";
            then?.Invoke(index);
        });
        if (!shown) Status = "This host has no dialogs";
    }
}
```
